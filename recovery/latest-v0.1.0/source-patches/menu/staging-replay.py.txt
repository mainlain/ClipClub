#!/usr/bin/env python3
"""Replay the final points-era compact menu changes into an explicit recovery tree.

Only ui/{index.html,style.css,app.js}, README.md and historical main-window size
are changed. Existing fork profile paths, ports, identity and title are preserved.
The original window diff is retained as 00-window.diff; its exact sizes may
already be present in the separately isolated fork.
"""
from pathlib import Path
import argparse,json,os,re
source=Path(__file__).resolve().parent
parser=argparse.ArgumentParser()
parser.add_argument('target',type=Path)
parser.add_argument('--skip-window',action='store_true',help='For replay verification from UI readbacks only')
args=parser.parse_args()
target=args.target.resolve(strict=True)
assert target!=Path('/Users/sean/Downloads/clipcrop'),'Refuse to modify current cryptocurrency project'
assert (target/'ui/app.js').is_file()
def patch_exact(path,diff):
    original=path.read_text()
    result=original
    parts=re.split(r'^@@[^\n]*\n',diff,flags=re.M)[1:]
    assert parts,'No diff hunks'
    for part in parts:
        old='';new=''
        for line in part.splitlines(keepends=True):
            if line.startswith((' ','-')):old+=line[1:]
            if line.startswith((' ','+')):new+=line[1:]
        if result.count(old)==1:
            result=result.replace(old,new,1)
        elif old not in result and new in result:
            print('Already applied hunk in',path.relative_to(target))
        else:
            raise RuntimeError(f'Historical hunk does not match uniquely in {path}: {old[:120]!r}')
    path.write_text(result)
# Preserve fork identity/profile safety. Only historical numeric window change.
if not args.skip_window:
    main=target/'src/main.rs';s=main.read_text()
    for old,new in [
        ('.with_inner_size(LogicalSize::new(1180.0, 830.0))','.with_inner_size(LogicalSize::new(480.0, 620.0))'),
        ('.with_min_inner_size(LogicalSize::new(740.0, 620.0))','.with_min_inner_size(LogicalSize::new(360.0, 480.0))'),
    ]:
        if old in s:s=s.replace(old,new,1)
        else:assert new in s,'Unexpected main-window size; preserve and resolve manually'
    if s==main.read_text():print('Historical 480×620 / 360×480 window already applied; fork identity preserved')
    else:main.write_text(s)
html=target/'ui/index.html'
if '<body data-page="home">' in html.read_text():
    assert 'id="page-home"' in html.read_text()
    assert 'The app opens in a compact, resizable window with four menu rows' in (target/'README.md').read_text()
    print('HTML / README menu transform already applied')
else:
    assert '<body>' in html.read_text()
    os.chdir(target)
    code=(source/'01-html-readme.py').read_text()
    exec(compile(code,str(source/'01-html-readme.py'),'exec'),{'__name__':'__historical_menu_replay__'})
(target/'ui/style.css').write_text((source/'02-style.css').read_text())
for step in json.loads((source/'steps.json').read_text()):
    patch_exact(target/step['target'],(source/step['file']).read_text())
print('Replayed complete historical compact-menu feature into',target)
