#!/usr/bin/env python3
"""Apply only recovered historical Roll changes after library-ratings restoration."""
from pathlib import Path
import argparse,json,subprocess,sys
parser=argparse.ArgumentParser()
parser.add_argument('target',type=Path)
parser.add_argument('--rustfmt',type=Path,default=Path('/Users/sean/Downloads/clipcrop/.build/rustup/toolchains/stable-aarch64-apple-darwin/bin/rustfmt'))
args=parser.parse_args()
base=Path(__file__).resolve().parent
stage=base.parent.parent.resolve()
target=args.target.resolve()
if not target.is_relative_to(stage):
 raise SystemExit('Recovery target must be inside this isolated staging directory')
log=[]
def run_fmt(path):
 subprocess.run([str(args.rustfmt),'--edition','2024',str(path)],check=True)
def clean_patch_artifacts(path):
 for suffix in ['.orig','.rej']:
  backup=Path(str(path)+suffix)
  if backup.exists():backup.unlink()
for action in json.loads((base/'actions.json').read_text()):
 print(action['kind'],action.get('path',action['file']),flush=True)
 if action['kind']=='script':
  subprocess.run([sys.executable,str(base/action['file'])],cwd=target,check=True)
  log.append(action|{'result':'applied'})
  continue
 path=target/action['path']
 path.parent.mkdir(parents=True,exist_ok=True)
 if action['kind']=='add':
  if path.exists():raise RuntimeError('Historical added file already exists: '+str(path))
  path.write_bytes((base/action['file']).read_bytes())
 else:
  # The original agents ran rustfmt between changes; repeat only when its
  # formatting is required to match a later recorded diff.
  original=path.read_bytes()
  patch=(base/action['file']).read_text()
  result=subprocess.run(['patch','--batch','--forward','-p1'],input=patch,text=True,cwd=target,capture_output=True)
  used_fmt=False
  if result.returncode:
   path.write_bytes(original);clean_patch_artifacts(path)
   if path.suffix=='.rs':
    run_fmt(path);used_fmt=True
    result=subprocess.run(['patch','--batch','--forward','-p1'],input=patch,text=True,cwd=target,capture_output=True)
   if result.returncode:
    path.write_bytes(original);clean_patch_artifacts(path)
    print(result.stdout,result.stderr,file=sys.stderr)
    raise RuntimeError('Historical patch did not apply: '+action['id'])
  clean_patch_artifacts(path)
  log.append(action|{'result':'applied','formatted_before_retry':used_fmt,'patch_output':result.stdout})
for path in sorted({target/a['path'] for a in json.loads((base/'actions.json').read_text()) if a.get('path','').endswith('.rs')}):
 run_fmt(path)
(base/'replay-log.json').write_text(json.dumps(log,indent=2)+'\n')
print('Restored all historical Roll changes.')
